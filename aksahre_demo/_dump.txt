  28|     return (
  29|         values.astype(str)
  30|         .str.strip()
  31|         .str.replace(r"\.0$", "", regex=True)
  32|         .str.replace(r"\.(SZ|SH|BJ)$", "", case=False, regex=True)

  38| def _monthly_market_data(quotes, adjusted=False):
  39|     required = {"股票代码", "日期", "收盘"}
  40|     if not adjusted:
  41|         required.update({"成交量", "换手率"})
  42|     missing_columns = required.difference(quotes.columns)
  43|     if missing_columns:
  44|         raise ValueError(f"行情数据缺少必要字段: {', '.join(sorted(missing_columns))}")
  45| 
  46|     columns = ["股票代码", "日期", "收盘"]
  47|     if not adjusted:
  48|         columns.extend(["成交量", "换手率"])
  49|     data = quotes.loc[:, columns].copy()
  50|     data["股票代码"] = _normalize_codes(data["股票代码"])
  51|     data["日期"] = pd.to_datetime(data["日期"], errors="raise").dt.normalize()
  52|     for column in columns[2:]:
  53|         data[column] = pd.to_numeric(data[column], errors="coerce")
  54|     data = data.dropna(subset=["股票代码", "日期", "收盘"])
  55|     data = data.loc[data["收盘"] > 0].copy()
  56|     if data.duplicated(["股票代码", "日期"]).any():
  57|         raise ValueError("行情数据存在重复的股票代码和日期记录")
  58|     data["月份"] = data["日期"].dt.to_period("M")
  59| 
  60|     if not adjusted:
  61|         valid_turnover = data.loc[data["换手率"] > 0, "换手率"]
  62|         if valid_turnover.empty:
  63|             raise ValueError("行情中没有有效换手率，无法估算流通市值")
  64|         turnover_scale = 10000 if valid_turnover.max() > 1.5 else 100
  65|         data["流通股本估算"] = (
  66|             data["成交量"] / data["换手率"] * turnover_scale
  67|         )
  68|         data["ln流通市值"] = np.log(
  69|             (data["收盘"] * data["流通股本估算"]).where(
  70|                 (data["收盘"] * data["流通股本估算"]) > 0
  71|             )
  72|         )
  73|         data.loc[~np.isfinite(data["ln流通市值"]), "ln流通市值"] = np.nan
  74| 
  75|     month_end = (
  76|         data.sort_values(["股票代码", "日期"])
  77|         .groupby(["股票代码", "月份"], as_index=False)
  78|         .tail(1)
  79|         .copy()
  80|     )
  81|     return month_end

  84| def calculate_monthly_ic(
  85|     factor,
  86|     unadjusted_quotes,
  87|     adjusted_quotes,
  88|     min_cross_section=MIN_CROSS_SECTION_SIZE,
  89| ):
  90|     required = {"股票代码", "可用日期", FACTOR_COLUMN}
  91|     missing_columns = required.difference(factor.columns)
  92|     if missing_columns:
  93|         raise ValueError(f"因子数据缺少必要字段: {', '.join(sorted(missing_columns))}")
  94| 
  95|     signals = factor.loc[:, ["股票代码", "可用日期", FACTOR_COLUMN]].copy()
  96|     signals["股票代码"] = _normalize_codes(signals["股票代码"])
  97|     signals["可用日期"] = pd.to_datetime(
  98|         signals["可用日期"], errors="raise"
  99|     ).dt.normalize()
 100|     signals[FACTOR_COLUMN] = pd.to_numeric(signals[FACTOR_COLUMN], errors="coerce")
 101|     signals = signals.replace([np.inf, -np.inf], np.nan).dropna(
 102|         subset=["股票代码", "可用日期", FACTOR_COLUMN]
 103|     )
 104|     signals = signals.sort_values(["股票代码", "可用日期"])
 105|     signals = signals.drop_duplicates(
 106|         ["股票代码", "可用日期"], keep="last"
 107|     )
 108| 
 109|     market = _monthly_market_data(unadjusted_quotes)
 110|     adjusted = _monthly_market_data(adjusted_quotes, adjusted=True)
 111|     market = market.rename(columns={"日期": "市值日期"})
 112|     adjusted_prices = adjusted.pivot(
 113|         index="月份", columns="股票代码", values="收盘"
 114|     )
 115|     all_months = pd.period_range(
 116|         adjusted_prices.index.min(), adjusted_prices.index.max(), freq="M"
 117|     )
 118|     adjusted_prices = adjusted_prices.reindex(all_months)
 119|     forward_returns = (
 120|         adjusted_prices.shift(-12).div(adjusted_prices).sub(1)
 121|     )
 122|     forward_returns.index.name = "月份"
 123|     forward_returns = (
 124|         forward_returns.rename_axis(columns="股票代码")
 125|         .reset_index()
 126|         .melt(id_vars="月份", var_name="股票代码", value_name="持有期收益")
 127|         .dropna(subset=["持有期收益"])
 128|     )
 129| 
 130|     market_months = sorted(market["月份"].unique())
 131|     results = []
 132|     for month in market_months:
 133|         month_end = market.loc[market["月份"] == month, "市值日期"].max()
 134|         as_of = signals.loc[signals["可用日期"] <= month_end]
 135|         if as_of.empty:
 136|             continue
 137|         as_of = (
 138|             as_of.sort_values(["股票代码", "可用日期"])
 139|             .drop_duplicates("股票代码", keep="last")
 140|         )
 141|         current_market = market.loc[market["月份"] == month]
 142|         current_returns = forward_returns.loc[forward_returns["月份"] == month]
 143|         cross_section = as_of.merge(
 144|             current_market[["股票代码", "月份", "ln流通市值"]],
 145|             on="股票代码",
 146|             how="inner",
 147|             validate="one_to_one",
 148|         ).merge(
 149|             current_returns[["股票代码", "月份", "持有期收益"]],
 150|             on=["股票代码", "月份"],
 151|             how="inner",
 152|             validate="one_to_one",
 153|         )
 154|         cross_section = cross_section.replace([np.inf, -np.inf], np.nan).dropna(
 155|             subset=[FACTOR_COLUMN, "ln流通市值", "持有期收益"]
 156|         )
 157|         if len(cross_section) < min_cross_section:
 158|             continue
 159| 
 160|         values = cross_section[FACTOR_COLUMN]
 161|         median = values.median()
 162|         mad = (values - median).abs().median()
 163|         if mad > 0 and np.isfinite(mad):
 164|             values = values.clip(
 165|                 lower=median - 5 * 1.4826 * mad,
 166|                 upper=median + 5 * 1.4826 * mad,
 167|             )
 168| 
 169|         design = np.column_stack(
 170|             [
 171|                 np.ones(len(cross_section)),
 172|                 cross_section["ln流通市值"].to_numpy(dtype=float),
 173|             ]
 174|         )
 175|         coefficients, _, _, _ = np.linalg.lstsq(
 176|             design, values.to_numpy(dtype=float), rcond=None
 177|         )
 178|         residuals = values.to_numpy(dtype=float) - design @ coefficients
 179|         residual_std = residuals.std(ddof=0)
 180|         if residual_std == 0 or not np.isfinite(residual_std):
 181|             continue
 182|         standardized = (residuals - residuals.mean()) / residual_std
 183|         returns = cross_section["持有期收益"].to_numpy(dtype=float)
 184|         rank_ic = pd.Series(standardized).rank(method="average").corr(
 185|             pd.Series(returns).rank(method="average")
 186|         )
 187|         pearson_ic = pd.Series(standardized).corr(pd.Series(returns))
 188|         results.append(
 189|             {
 190|                 "截面月份": str(month),
 191|                 "截面日期": month_end.strftime("%Y-%m-%d"),
 192|                 "预测月份": str(month + FORWARD_HORIZON_MONTHS),
 193|                 "股票数": len(cross_section),
 194|                 "Rank IC": rank_ic,
 195|                 "Pearson IC": pearson_ic,
 196|             }
 197|         )
 198| 
 199|     return pd.DataFrame(
 200|         results,
 201|         columns=[
 202|             "截面月份",
 203|             "截面日期",
 204|             "预测月份",
 205|             "股票数",
 206|             "Rank IC",
 207|             "Pearson IC",
 208|         ],
 209|     )

 212| def _statistics(values):
 213|     values = values.dropna()
 214|     if values.empty:
 215|         return {
 216|             "mean": float("nan"),
 217|             "std": float("nan"),
 218|             "ir": float("nan"),
 219|             "positive_ratio": float("nan"),
 220|             "t_stat": float("nan"),
 221|         }
 222|     mean = values.mean()
 223|     standard_deviation = values.std(ddof=1) if len(values) > 1 else float("nan")
 224|     standard_error = standard_deviation / np.sqrt(len(values))
 225|     return {
 226|         "mean": mean,
 227|         "std": standard_deviation,
 228|         "ir": mean / standard_deviation
 229|         if np.isfinite(standard_deviation) and standard_deviation > 0
 230|         else float("nan"),
 231|         "positive_ratio": (values > 0).mean(),
 232|         "t_stat": mean / standard_error
 233|         if np.isfinite(standard_error) and standard_error > 0
 234|         else float("nan"),
 235|     }

 148| def _beta_continued_fraction(a, b, x):
 149|     maximum_iterations = 200
 150|     epsilon = 3e-14
 151|     tiny = 1e-300
 152|     qab = a + b
 153|     qap = a + 1
 154|     qam = a - 1
 155|     c = 1.0
 156|     d = 1 - qab * x / qap
 157|     if abs(d) < tiny:
 158|         d = tiny
 159|     d = 1 / d
 160|     result = d
 161| 
 162|     for iteration in range(1, maximum_iterations + 1):
 163|         m2 = 2 * iteration
 164|         numerator = iteration * (b - iteration) * x / (
 165|             (qam + m2) * (a + m2)
 166|         )
 167|         d = 1 + numerator * d
 168|         if abs(d) < tiny:
 169|             d = tiny
 170|         c = 1 + numerator / c
 171|         if abs(c) < tiny:
 172|             c = tiny
 173|         d = 1 / d
 174|         result *= d * c
 175| 
 176|         numerator = -(a + iteration) * (qab + iteration) * x / (
 177|             (a + m2) * (qap + m2)
 178|         )
 179|         d = 1 + numerator * d
 180|         if abs(d) < tiny:
 181|             d = tiny
 182|         c = 1 + numerator / c
 183|         if abs(c) < tiny:
 184|             c = tiny
 185|         d = 1 / d
 186|         delta = d * c
 187|         result *= delta
 188|         if abs(delta - 1) < epsilon:
 189|             return result
 190| 
 191|     raise ArithmeticError("Student-t p 值计算未收敛")

 194| def _regularized_incomplete_beta(x, a, b):
 195|     if x <= 0:
 196|         return 0.0
 197|     if x >= 1:
 198|         return 1.0
 199|     front = math.exp(
 200|         math.lgamma(a + b)
 201|         - math.lgamma(a)
 202|         - math.lgamma(b)
 203|         + a * math.log(x)
 204|         + b * math.log1p(-x)
 205|     )
 206|     if x < (a + 1) / (a + b + 2):
 207|         return front * _beta_continued_fraction(a, b, x) / a
 208|     return 1 - front * _beta_continued_fraction(b, a, 1 - x) / b

 211| def two_sided_t_pvalue(t_statistic, degrees_of_freedom):
 212|     if degrees_of_freedom < 1 or not np.isfinite(t_statistic):
 213|         return float("nan")
 214|     x = degrees_of_freedom / (degrees_of_freedom + float(t_statistic) ** 2)
 215|     return _regularized_incomplete_beta(x, degrees_of_freedom / 2, 0.5)

 218| def _newey_west_mean_standard_error(values, max_lags=HAC_MAX_LAGS):
 219|     values = np.asarray(values, dtype=float)
 220|     if values.ndim != 1 or not np.isfinite(values).all():
 221|         raise ValueError("HAC 系数序列必须是一维有限数值")
 222|     if max_lags < 0:
 223|         raise ValueError("HAC 滞后期数不能为负数")
 224|     observations = len(values)
 225|     if observations <= max_lags:
 226|         raise ValueError(
 227|             f"HAC 滞后期数 {max_lags} 必须小于系数观测数 {observations}"
 228|         )
 229| 
 230|     residuals = values - values.mean()
 231|     long_run_variance = float(np.dot(residuals, residuals) / observations)
 232|     for lag in range(1, max_lags + 1):
 233|         bartlett_weight = 1 - lag / (max_lags + 1)
 234|         autocovariance = float(
 235|             np.dot(residuals[lag:], residuals[:-lag]) / observations
 236|         )
 237|         long_run_variance += 2 * bartlett_weight * autocovariance
 238| 
 239|     variance_of_mean = long_run_variance / observations
 240|     variance_of_mean *= observations / (observations - 1)
 241|     if variance_of_mean < 0 and np.isclose(variance_of_mean, 0):
 242|         variance_of_mean = 0.0
 243|     if variance_of_mean < 0:
 244|         raise ArithmeticError("Newey-West HAC 方差估计为负")
 245|     return float(np.sqrt(variance_of_mean))

 248| def _coefficient_summary(values, max_lags=HAC_MAX_LAGS):
 249|     values = np.asarray(values, dtype=float)
 250|     mean = float(values.mean())
 251|     standard_error = _newey_west_mean_standard_error(values, max_lags=max_lags)
 252|     t_stat = (
 253|         mean / standard_error
 254|         if np.isfinite(standard_error) and standard_error > 0
 255|         else float("nan")
 256|     )
 257|     return {
 258|         "mean": mean,
 259|         "standard_error": standard_error,
 260|         "t_stat": t_stat,
 261|         "p_value": two_sided_t_pvalue(t_stat, len(values) - 1),
 262|     }
