 150| def _clean_cross_section(cross_section):
 151|     """截面清洗：MAD(5 倍) 去极值 → 对数流通市值回归中性化 → Z-score。"""
 152|     values = cross_section[FACTOR_COLUMN].to_numpy(dtype="float64")
 153|     median = np.median(values)
 154|     mad = np.median(np.abs(values - median))
 155|     if mad > 0 and np.isfinite(mad):
 156|         values = np.clip(values, median - 5 * 1.4826 * mad, median + 5 * 1.4826 * mad)
 157| 
 158|     design = np.column_stack(
 159|         [np.ones(len(values)), cross_section["ln流通市值"].to_numpy(dtype="float64")]
 160|     )
 161|     coefficients, _, _, _ = np.linalg.lstsq(design, values, rcond=None)
 162|     residuals = values - design @ coefficients
 163|     residual_std = residuals.std(ddof=0)
 164|     if residual_std == 0 or not np.isfinite(residual_std):
 165|         cleaned = np.zeros(len(residuals))
 166|     else:
 167|         cleaned = (residuals - residuals.mean()) / residual_std
 168| 
 169|     cross_section = cross_section.copy()
 170|     cross_section["因子截尾值"] = values
 171|     cross_section["因子清洗值"] = cleaned
 172|     return cross_section
