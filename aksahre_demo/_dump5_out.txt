 102| def to_single_quarter(cumulative):
 103|     """累计值 → 单季度值：Q1 取累计值本身，Q2/Q3/Q4 与上一季度（网格上一行）差分。
 104| 
 105|     网格连续，因此 shift(1) 即上一季度；上一季度缺失时结果为 NaN（不做插值）。
 106|     """
 107|     first_quarter = (cumulative.index % 4) == 1
 108|     single = cumulative - cumulative.shift(1)
 109|     return single.mask(first_quarter, cumulative, axis=0)
