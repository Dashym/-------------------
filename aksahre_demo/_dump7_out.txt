 216| def next_trading_day(notice_dates, trading_days):
 217|     """公告日（NaT 表示缺失）→ 严格晚于公告日的第一个交易日；NaT 保持缺失。"""
 218|     values = np.asarray(notice_dates, dtype="datetime64[ns]")
 219|     shape = values.shape
 220|     flat = values.reshape(-1).astype("datetime64[D]")
 221|     days = np.asarray(trading_days, dtype="datetime64[D]")
 222|     result = np.full(flat.shape, np.datetime64("NaT"), dtype="datetime64[ns]")
 223|     valid = ~np.isnat(flat)
 224|     if valid.any():
 225|         positions = np.searchsorted(days, flat[valid], side="right")
 226|         inside = positions < len(days)
 227|         target = np.nonzero(valid)[0][inside]
 228|         result[target] = days[positions[inside]].astype("datetime64[ns]")
 229|     return result.reshape(shape)

 243| def build_month_end_panel(detail, pool, cross_section_dates):
 244|     """按月末截面日取"已公告的最新因子值"，生成截面面板。"""
 245|     usable = detail.dropna(subset=[FACTOR_COLUMN, "可用日期"])
 246|     usable = usable.sort_values(["股票代码", "报告期"], kind="mergesort")
 247|     frames = []
 248|     for as_of in cross_section_dates:
 249|         available = usable.loc[usable["可用日期"] <= as_of]
 250|         if available.empty:
 251|             continue
 252|         latest = available.drop_duplicates("股票代码", keep="last").copy()
 253|         latest = latest.merge(pool, on="股票代码", how="inner", validate="many_to_one")
 254|         latest = latest.loc[latest["上市日"] <= as_of]
 255|         latest["可用日期"] = as_of
 256|         latest["截面月份"] = as_of.to_period("M").strftime("%Y-%m")
 257|         frames.append(latest)
 258|     if not frames:
 259|         raise ValueError("没有任何月末截面可用，请检查数据区间与公告日期")
 260|     panel = pd.concat(frames, ignore_index=True)
 261|     return panel.sort_values(["可用日期", "股票代码"], kind="mergesort").reset_index(drop=True)
